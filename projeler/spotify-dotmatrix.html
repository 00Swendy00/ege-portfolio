Proje hakkında

Bilgisayarda çalışan Python betiği, açık pencerelerin başlıklarını kontrol ederek müzik bilgisi olabilecek metni seri bağlantıyla Arduino Uno'ya iletir. Arduino, gelen metni MAX7219 LED matrisler üzerinde kaydırır. Bu örnek Spotify'ın resmî Web API'sini kullanmaz; masaüstündeki pencere başlıklarını okur.

Kullanılan araçlar ve bileşenler

  • Arduino Uno
  • MAX7219 sürücülü 4 adet 8×8 LED Dot Matrix modülü
  • USB bağlantısı ve jumper kablolar
  • Python, pyserial ve pygetwindow
  • Arduino IDE, MD_Parola ve MD_MAX72XX kütüphaneleri

spotify-dotmatrix.ino
#include <MD_Parola.h>
#include <MD_MAX72xx.h>
#include <SPI.h>

#define HARDWARE_TYPE MD_MAX72XX::FC16_HW
#define MAX_DEVICES 4
#define DATA_PIN 11
#define CS_PIN 10
#define CLK_PIN 13

MD_Parola ekran(HARDWARE_TYPE, DATA_PIN, CLK_PIN, CS_PIN, MAX_DEVICES);
String mesaj = "Spotify hazir";

void setup() {
  Serial.begin(9600);
  Serial.setTimeout(100);
  ekran.begin();
  ekran.setIntensity(3);
  ekran.displayClear();
  ekran.displayText(mesaj.c_str(), PA_LEFT, 65, 0, PA_SCROLL_LEFT, PA_SCROLL_LEFT);
}

void loop() {
  if (Serial.available()) {
    String yeni = Serial.readStringUntil('\n');
    yeni.trim();
    if (yeni.length() > 0) {
      mesaj = yeni;
      ekran.displayClear();
      ekran.displayText(mesaj.c_str(), PA_LEFT, 65, 0, PA_SCROLL_LEFT, PA_SCROLL_LEFT);
    }
  }
  ekran.displayAnimate();
}

spotify-pencere-basligi.py
# Windows'ta pencere basligini okuyup Arduino'ya seri porttan aktarir.
# Kurulum: pip install pyserial pygetwindow
# Not: Spotify Web API entegrasyonu degildir.
import time
import serial
import pygetwindow as gw

PORT = "COM9"  # Aygit Yoneticisi'ndeki Arduino COM portuyla degistirin.
BAUD = 9600


def ascii_cevir(metin):
    return metin.translate(str.maketrans("çğıöşüÇĞİÖŞÜ", "cgiosuCGIOSU"))


def baslik_bul():
    # Masaustunde Spotify muzik penceresinin basligi gorunebilir.
    # Baska programlarin basliklarini sarki sanmamak icin filtre koyduk.
    yasak = ("Google Chrome", "Microsoft Edge", "Visual Studio", "Discord", "Opera", "Administrator", "Arduino")
    for title in gw.getAllTitles():
        if title and " - " in title and not any(x.lower() in title.lower() for x in yasak):
            return title
    return None


with serial.Serial(PORT, BAUD, timeout=1) as arduino:
    time.sleep(2)
    onceki = None
    while True:
        baslik = baslik_bul()
        if baslik and baslik != onceki:
            onceki = baslik
            arduino.write((ascii_cevir(baslik) + "\n").encode("utf-8"))
            print("Gonderildi:", baslik)
        time.sleep(2)
